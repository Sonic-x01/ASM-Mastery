Assembly مع C، الـ Macros، وبرامج Windows حقيقية

في الحقيقة نادرًا ما برنامج بيبقى Assembly 100%: الـ malware بيخلط C مع inline assembly (anti-debug، syscalls مباشرة)، والـ shellcode بيتكتب Assembly وبيتنادي من loader بـ C. المرحلة دي بتوصّل كل ده: إزاي تنادي C من Assembly والعكس، الـ inline asm، الـ NASM macros، وبرامج Windows x64 بتكلّم الـ WinAPI مباشرة.

C ↔ Assembly global · extern · link

📤

Assembly بيصدّر دالة لـ C

في الـ .asm: global my_func، والدالة تحترم الـ calling convention. في C: long my_func(long a, long b); وتناديها عادي. وبعدين: nasm -f elf64 f.asm + gcc main.c f.o.

📥

Assembly بينادي C (أو libc)

extern printf، وبعدين call printf بعد ما تحط الـ args وتظبط الـ alignment (rsp مضاعف 16) وal = عدد الـ xmm args. الـ entry تبقى main مش _start وgcc يعمل الـ link.

; build: nasm -f elf64 printf_from_asm.asm && gcc -no-pie printf_from_asm.o -o p && ./p default rel extern printf section .data fmt: db "%s has %d bytes, key=0x%lx", 10, 0 name: db "payload.bin", 0 section .text global main main: push rbp ; entry rsp is 8 mod 16 -> push makes it 16-aligned lea rdi, [fmt] ; arg1: format lea rsi, [name] ; arg2: %s mov edx, 4096 ; arg3: %d mov rcx, 0xDEADBEEF ; arg4: %lx xor eax, eax ; al = 0 vector registers (varargs rule) call printf xor eax, eax ; return 0 from main pop rbp ret

المثال ده بيتعمله assemble وبيتأكد إنه سليم، بس مش بيتشغل هنا (Compiler Explorer بيربط NASM من غير libc). جرّبه في WSL بالأوامر اللي في أوله. والعكس — دالة Assembly بتتنادى من C — ده شغّال جوه الصفحة (بصيغة GAS Intel اللي gcc بيفهمها):

#include /* A whole function written in assembly, called from C (System V ABI). */ __asm__( ".intel_syntax noprefix\n" ".text\n" ".globl xor_crypt\n" "xor_crypt:\n" /* void xor_crypt(char *buf, long n, char key) */ " test rsi, rsi\n" " jz 2f\n" "1: xor byte ptr [rdi], dl\n" /* local numeric labels: 1f / 1b */ " inc rdi\n" " dec rsi\n" " jnz 1b\n" "2: ret\n" ".att_syntax prefix\n" ); void xor_crypt(char *buf, long n, char key); int main(void) { char secret[] = "beacon.example.net"; xor_crypt(secret, sizeof secret - 1, 0x42); printf("encrypted: "); for (unsigned i = 0; i < sizeof secret - 1; i++) printf("%02x", (unsigned char)secret[i]); xor_crypt(secret, sizeof secret - 1, 0x42); printf("\ndecrypted: %s\n", secret); return 0; }

Inline Assembly GCC extended asm

لما تحتاج instruction واحدة جوه كود C (cpuid، rdtsc، syscall)، gcc عنده extended asm: بتقوله الـ inputs والـ outputs والـ registers اللي بتبوّظها، وهو بيظبط الباقي. ده كود anti-VM حقيقي الشكل: بيقرا الـ vendor والـ hypervisor bit:

#include #include int main(void) { unsigned int a, b, c, d; char vendor[13]; __asm__ volatile ("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(d) /* outputs */ : "a"(0)); /* input: leaf 0 */ memcpy(vendor + 0, &b, 4); /* the vendor string is EBX, EDX, ECX */ memcpy(vendor + 4, &d, 4); memcpy(vendor + 8, &c, 4); vendor[12] = 0; printf("vendor: %s\n", vendor); __asm__ volatile ("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(d) : "a"(1)); printf("hypervisor bit (ecx[31]) = %u\n", c >> 31); return 0; }

(المخرجات بتعتمد على الـ server اللي Compiler Explorer شغال عليه — وغالبًا هتلاقي الـ hypervisor bit = 1، لأنه VM في الـ cloud! بالظبط اللي الـ malware بيدور عليه.)

🔬
MSVC x64 مفيهوش inline asm!

MSVC بيسمح بـ __asm { } في الـ 32-bit بس. في الـ x64 الـ malware authors بيستخدموا intrinsics (__cpuid، __rdtsc، __readgsqword(0x60) = PEB!، _ReturnAddress) أو ملفات .asm منفصلة (MASM) — زي الـ direct syscall stubs (SysWhispers، Hell's Gate). لو لقيت دالة صغيرة كلها mov r10, rcx + mov eax, N + syscall جوه EXE عادي ← direct syscalls.

الـ NASM Macros preprocessor

NASM عنده preprocessor قوي. الـ shellcode والـ malware المكتوب بـ Assembly بيستخدمه بكثرة (مثلًا macro بيحسب hash الـ API وقت الـ assembly). المهم كـ reverser: الـ macros بتختفي في الـ binary — اللي بتشوفه هو الـ instructions اللي اتولّدت.

%define STDOUT 1 ; text substitution %define SYS_WRITE 1 %macro print_literal 1 ; a macro with 1 parameter jmp %%skip ; %% = label unique to each expansion %%text: db %1, 10 %%len equ $ - %%text %%skip: mov eax, SYS_WRITE mov edi, STDOUT lea rsi, [rel %%text] mov edx, %%len syscall %endmacro struc conn ; NASM struct definition = offsets .port: resw 1 .flags: resw 1 .addr: resd 1 endstruc ; conn_size = 8 section .text global _start _start: print_literal "macro expanded #1" print_literal "macro expanded #2" %assign i 1 %rep 3 ; repeated at assemble time mov edi, i call print_int call print_nl %assign i i*10 %endrep mov edi, conn.addr ; = 4 (struc offsets are constants) call print_int call print_nl mov edi, conn_size ; = 8 call print_int call print_nl xor edi, edi call exit
💡
ملاحظة على المثال

الـ macro بيحط الـ string جوه الـ .text ويقفز فوقها (jmp %%skip) — بالظبط الحيلة اللي الـ shellcode بيستخدمها عشان يبقى قطعة واحدة. المعمل بتاعنا مش بيدعم الـ macros ولا البيانات جوه الكود، فالمثال ده بيتشغل على Compiler Explorer بس. اضغط ⚙ Bytes وشوف الـ string مدفونة وسط الـ instructions.

برنامج Windows x64 بـ NASM WinAPI

على Windows مفيش syscalls موثّقة للبرامج — بتنادي الـ API (DLLs) وتحترم الـ Microsoft x64 convention: rcx, rdx, r8, r9، 32 byte shadow space، وrsp مضاعف 16 قبل كل call. ده برنامج console كامل (اتعمله assemble هنا بـ -f win64، ونسخة مطابقة منه اتشغّلت فعلًا على Windows 10 وطبعت السطر):

; nasm -f win64 win_hello.asm -o win_hello.obj ; gcc -nostartfiles -Wl,-e,start win_hello.obj -o win_hello.exe -lkernel32 (MinGW) ; link /subsystem:console /entry:start win_hello.obj kernel32.lib (MSVC) default rel extern GetStdHandle extern WriteFile extern ExitProcess section .data msg: db "Hello from Windows x64!", 13, 10 msg_len equ $ - msg section .bss written: resd 1 section .text global start start: sub rsp, 40 ; 32 shadow space + 8 to realign (entry rsp = 8 mod 16) mov ecx, -11 ; STD_OUTPUT_HANDLE call GetStdHandle ; rax = console handle mov rcx, rax ; 1: hFile lea rdx, [msg] ; 2: lpBuffer mov r8d, msg_len ; 3: nNumberOfBytesToWrite lea r9, [written] ; 4: lpNumberOfBytesWritten mov qword [rsp+32], 0 ; 5: lpOverlapped = NULL (above the shadow space) call WriteFile xor ecx, ecx ; exit code 0 call ExitProcess ; a GUI message box: link with -luser32 / user32.lib as well default rel extern MessageBoxA extern ExitProcess section .data title: db "ASM Mastery", 0 text: db "Hello from NASM!", 0 section .text global start start: sub rsp, 40 xor ecx, ecx ; hWnd = NULL lea rdx, [text] ; lpText lea r8, [title] ; lpCaption mov r9d, 0x40 ; uType = MB_ICONINFORMATION call MessageBoxA xor ecx, ecx call ExitProcess
🔬
نفس البرنامج في IDA / x64dbg

الـ call MessageBoxA هتظهر call cs:__imp_MessageBoxA (IDA) أو call qword ptr ds:[<&MessageBoxA>] (x64dbg): نداء غير مباشر عن طريق جدول الـ imports (الـ IAT). الـ Imports tab لوحده بيقولك كتير عن أي sample: VirtualAlloc + WriteProcessMemory + CreateRemoteThread = injection. هنفصّل الـ IAT في مرحلة PE.

في الـ .asm عايز C يقدر ينادي decrypt. محتاج: extern decrypt global decrypt section .decrypt %define decrypt global = صدّر الاسم. extern = استورد اسم معرّف في مكان تاني. واحد للتصدير وواحد للاستيراد. في برنامج Windows x64 بـ NASM نسيت sub rsp, 40 قبل call WriteFile. إيه اللي ممكن يحصل؟ ولا حاجة خطأ في الـ assembly البرنامج هيبقى أبطأ الـ API هتكتب الـ shadow space فوق عنوان الرجوع / بياناتك، والـ stack مش aligned — crash أو سلوك عشوائي الـ callee مسموحله يكتب في الـ 32 byte اللي فوق عنوان رجوعه، وبعض الدوال بتستخدم SSE instructions بتعمل crash لو الـ stack مش aligned. الـ shadow space ملك الـ callee. في GCC inline asm، الـ clobber "memory" معناه: الـ asm ممكن يقرا/يكتب ذاكرة — متخليش الـ compiler يحتفظ بقيم ذاكرة في registers عبره بيحجز ذاكرة بيمسح الذاكرة الـ asm مبيلمسش الذاكرة من غيره الـ optimizer ممكن يفترض إن الذاكرة متغيرتش ويستخدم قيمة قديمة. الـ clobbers = حاجات الـ asm بيبوّظها. EXE مكتوب بـ C على Windows x64 (MSVC) فيه دالة: mov r10, rcx / mov eax, 18h / syscall / ret. ده: inline asm عادي كود ntdll طبيعي direct syscall stub — الـ malware بينادي الـ kernel من غير ما يعدّي على ntdll (وبيتخطى hooks الـ EDR) anti-VM الشكل ده مكانه الطبيعي ntdll.dll. وجوده في الـ EXE نفسه = تقنية evasion (SysWhispers/Hell's Gate). 0x18 = رقم الـ syscall (SSN). MSVC x64 مفيهوش inline asm، والشكل ده نسخة من stubs بتاعة dll معيّنة… NASM: %rep 4 / nop / %endrep. في الـ binary هتلاقي: loop بتتكرر 4 مرات 4 instructions nop ورا بعض (0x90 × 4) — الـ macro بيتفك وقت الـ assembly instruction rep ولا حاجة الـ preprocessor بيكرر النص قبل الـ assembly. مفيش loop وقت التشغيل. %rep شغالة إمتى: assemble time ولا run time؟ __readgsqword(0x60) في كود C (MSVC x64) بيرجع: الـ TEB الـ stack canary الـ ImageBase عنوان الـ PEB intrinsic بيتحول لـ mov rax, gs:[60h]. أشهر سطر في كود الـ anti-debug والـ API resolving. gs:[60h] على Windows x64… كمّل الدالة count_upper بالـ Assembly (GAS Intel syntax جوه __asm__): بتاخد string في rdi وبترجع عدد الحروف الكبيرة (A–Z) في rax. الـ main بتطبع النتيجة لـ 3 strings:
